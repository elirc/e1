export function createBooking(input: {
  eventTypeId?: string;
  startTime?: string;
  attendeeEmail?: string;
  attendeeName?: string;
}) {
  const eventType = getEventType(input.eventTypeId ?? "evt_strategy_sync");

  if (!eventType) {
    throw new Error("Event type not found.");
  }

  const requestedStart = new Date(input.startTime ?? new Date().toISOString());
  const requestedEnd = new Date(
    requestedStart.getTime() + eventType.durationMinutes * 60_000,
  );

  const conflict = getBookings().some((booking) => {
    if (
      booking.eventTypeId !== eventType.id ||
      (booking.status !== "confirmed" && booking.status !== "pending")
    ) {
      return false;
    }

    const existingStart = new Date(booking.startTime).getTime();
    const existingEnd = new Date(booking.endTime).getTime();

    return requestedStart.getTime() < existingEnd && requestedEnd.getTime() > existingStart;
  });

  if (conflict) {
    throw new Error("Requested slot conflicts with an active booking.");
  }

  return {
    booking: {
      id: `book_${crypto.randomUUID()}`,
      teamId: eventType.teamId,
      eventTypeId: eventType.id,
      organizerId: eventType.organizerId,
      status: "confirmed" as const,
      startTime: requestedStart.toISOString(),
      endTime: requestedEnd.toISOString(),
      timezone: eventType.timezone,
      title: `${eventType.title} with ${input.attendeeName ?? "Guest"}`,
      locationType: eventType.locationType,
      locationLabel: eventType.locationLabel,
      confirmationToken: crypto.randomUUID(),
      cancellationReason: null,
      rescheduledFromBookingId: null,
      guestCanCancel: true,
      guestCanReschedule: true,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      deletedAt: null,
    },
    attendee: {
      email: input.attendeeEmail ?? "guest@example.com",
      name: input.attendeeName ?? "Guest Booker",
    },
  };
}

export function confirmBooking(bookingId: string) {
  const booking = getBookingById(bookingId);
  if (!booking) {
    throw new Error("Booking not found.");
  }

  if (!canTransitionBooking(booking.status, "confirmed")) {
    return booking;
  }

  return transitionBooking(booking, "confirmed");
}

export function cancelBooking(bookingId: string, reason: string) {
  const booking = getBookingById(bookingId);
  if (!booking) {
    throw new Error("Booking not found.");
  }

  return transitionBooking(booking, "canceled", reason);
}

export function rescheduleBooking(bookingId: string, startTime: string) {
  const booking = getBookingById(bookingId);
  if (!booking) {
    throw new Error("Booking not found.");
  }

  const previous = transitionBooking(
    booking,
    "canceled",
    "Rescheduled by guest or host",
  );

  const eventType = getEventType(booking.eventTypeId);

  if (!eventType) {
    throw new Error("Event type not found.");
  }

  return {
    previous,
    replacement: {
      ...booking,
      id: `book_${crypto.randomUUID()}`,
      status: "confirmed" as const,
      startTime,
      endTime: new Date(
        new Date(startTime).getTime() + eventType.durationMinutes * 60_000,
      ).toISOString(),
      rescheduledFromBookingId: booking.id,
      updatedAt: new Date().toISOString(),
    },
  };
}

export function getBookingDetail(bookingId: string) {
  const booking = getBookingById(bookingId);
  const attendeeList = getBookingAttendees().filter(
    (attendee) => attendee.bookingId === bookingId,
  );

  if (!booking) {
    return null;
  }

  return {
    booking,
    attendees: attendeeList,
    eventType: getEventType(booking.eventTypeId),
  };
}

export function getBookingDetailByToken(token: string) {
  const booking = getBookingByToken(token);

  if (!booking) {
    return null;
  }

  return {
    booking,
    attendees: getBookingAttendees().filter(
      (attendee) => attendee.bookingId === booking.id,
    ),
  };
}
